```python
from sqlalchemy import Column, Integer, String, DateTime, Float
from datetime import datetime
from database import Base

class ParseConfig(Base):
    __tablename__ = "parse_configs"

    id = Column(Integer, primary_key=True, index=True)
    name = Column(String)
    base_url = Column(String)
    priority = Column(Integer, default=0)
    status = Column(String, default="active")  # active / inactive
    # 新增：两轮测速延迟（秒），None 表示未测试或失败
    latency1 = Column(Float, nullable=True)  # 第一轮 httpx 延迟
    latency2 = Column(Float, nullable=True)  # 第二轮 Playwright 延迟
    created_at = Column(DateTime, default=datetime.utcnow)
```

- [ ] **Step 1: 在 parse_configs.py 增加测速端点**

```python
# backend/api/parse_configs.py
from fastapi import APIRouter, Depends, HTTPException
from sqlalchemy.orm import Session
from database import get_db
from models.parse_config import ParseConfig
from schemas.parse_config import ParseConfigCreate, ParseConfigUpdate, ParseConfigResponse
import asyncio

router = APIRouter(prefix="/api/parse-configs", tags=["parse-configs"])

# ... 现有端点保持不变 ...

@router.post("/speed-test/{config_id}", response_model=ParseConfigResponse)
async def speed_test_single(config_id: int, db: Session = Depends(get_db)):
    """对单个解析服务进行两轮测速"""
    config = db.query(ParseConfig).get(config_id)
    if not config:
        raise HTTPException(status_code=404, detail="Config not found")

    from crawlers.parse_config_crawler import test_parse_config, test_parse_config_browser, build_parse_url

    # 第一轮：httpx 快速测试
    available1, latency1 = await test_parse_config(config.base_url)

    # 第二轮：Playwright 浏览器测试
    test_video_url = "https://v.qq.com/x/cover/mzc00200x8gfhok/m4100c4yead.html"
    full_url = build_parse_url(config.base_url, test_video_url)
    available2, latency2, _ = await test_parse_config_browser(full_url, timeout=30.0)

    # 更新数据库
    config.latency1 = latency1 if available1 else None
    config.latency2 = latency2 if available2 else None
    db.commit()
    db.refresh(config)

    return config


@router.post("/speed-test-all", response_model=list[ParseConfigResponse])
async def speed_test_all(db: Session = Depends(get_db)):
    """对所有解析服务进行两轮测速"""
    configs = db.query(ParseConfig).filter_by(status="active").all()

    from crawlers.parse_config_crawler import test_parse_config, test_parse_config_browser, build_parse_url

    test_video_url = "https://v.qq.com/x/cover/mzc00200x8gfhok/m4100c4yead.html"

    async def test_one(config):
        # 第一轮：httpx 快速测试
        available1, latency1 = await test_parse_config(config.base_url)

        # 第二轮：Playwright 浏览器测试
        full_url = build_parse_url(config.base_url, test_video_url)
        available2, latency2, _ = await test_parse_config_browser(full_url, timeout=30.0)

        config.latency1 = latency1 if available1 else None
        config.latency2 = latency2 if available2 else None
        return config

    # 并发测速，使用信号量控制并发为 3
    semaphore = asyncio.Semaphore(3)

    async def test_with_semaphore(config):
        async with semaphore:
            return await test_one(config)

    tasks = [test_with_semaphore(c) for c in configs]
    results = await asyncio.gather(*tasks)

    db.commit()
    return results
```

```vue
<!-- frontend/src/views/Settings.vue -->
<template>
  <div class="page-container">
    <header class="page-header">

      <div class="table-wrapper">
        <el-table v-loading="loading" :data="configs" stripe>
          <el-table-column prop="id" label="#" width="60" align="center" />
          <el-table-column prop="name" label="名称" />
          <el-table-column prop="base_url" label="链接">
            <template #default="{ row }">
              <a :href="row.base_url" target="_blank" class="parser-link">
                {{ row.base_url }}
              </a>
            </template>
          </el-table-column>
          <el-table-column label="状态" width="160" align="center">
            <template #default="{ row }">
              <span v-if="row.speedTesting" class="speed-testing">
                <el-icon class="is-loading"><Loading /></el-icon> 测速中
              </span>
              <span v-else class="speed-status">
                {{ formatLatency(row.latency1) }} / {{ formatLatency(row.latency2) }}
              </span>
            </template>
          </el-table-column>
          <el-table-column label="操作" width="140" align="center">
            <template #default="{ row }">
              <el-button size="small" @click="handleCopy(row.base_url)">复制</el-button>
              <el-button size="small" type="primary" @click="handleSpeedTest(row)" :loading="row.speedTesting">
                测速
              </el-button>
            </template>
          </el-table-column>
        </el-table>
      </div>
    </div>
  </div>
</template>

<script setup>
import { ref, onMounted } from 'vue'
import { ElMessage } from 'element-plus'
import { Loading } from '@element-plus/icons-vue'
import { getParseConfigs, speedTestParser, speedTestAllParsers } from '../api'

const configs = ref([])
const loading = ref(false)
const speedingAll = ref(false)

onMounted(() => {
  loadConfigs()
})

const loadConfigs = async () => {
  loading.value = true
  try {
    const { data } = await getParseConfigs()
    configs.value = data.map(c => ({ ...c, speedTesting: false }))
  } catch (err) {
    ElMessage.error('加载解析服务失败')
  } finally {
    loading.value = false
  }
}

const formatLatency = (latency) => {
  if (latency === null || latency === undefined) return '✗'
  return `${latency}s`
}

const handleCopy = async (url) => {
  try {
    await navigator.clipboard.writeText(url)
    ElMessage.success('链接已复制')
  } catch {
    ElMessage.error('复制失败')
  }
}

const handleSpeedTest = async (config) => {
  config.speedTesting = true
  try {
    const { data } = await speedTestParser(config.id)
    config.latency1 = data.latency1
    config.latency2 = data.latency2
    ElMessage.success('测速完成')
  } catch (err) {
    ElMessage.error('测速失败')
  } finally {
    config.speedTesting = false
  }
}

const handleSpeedTestAll = async () => {
  speedingAll.value = true
  configs.value.forEach(c => c.speedTesting = true)
  try {
    const { data } = await speedTestAllParsers()
    data.forEach((item, index) => {
      if (configs.value[index]) {
        configs.value[index].latency1 = item.latency1
        configs.value[index].latency2 = item.latency2
        configs.value[index].speedTesting = false
      }
    })
    ElMessage.success('全部测速完成')
  } catch (err) {
    ElMessage.error('测速失败')
  } finally {
    speedingAll.value = false
    configs.value.forEach(c => c.speedTesting = false)
  }
}
</script>
```
